auto-dev-stop.sh ​

bash
#!/bin/bash
# Auto-Dev Stop Hook — TDD-based autonomous loop controller
# Integrates with existing tdd-guide + quality-gate
#
# Inspired by Director Mode Lite's auto-loop-stop.sh
# Adapted to work with the existing ECC ecosystem
#
# Input: Claude Code Stop hook format (no stdin needed)
# Output: JSON with "decision": "block" to continue, or exit 0 to stop

set -euo pipefail

STATE_DIR=".auto-dev"
CHECKPOINT_FILE="$STATE_DIR/checkpoint.json"
STOP_FILE="$STATE_DIR/stop"
ITERATION_FILE="$STATE_DIR/iteration.txt"

# No active loop → allow normal exit
if [[ ! -f "$CHECKPOINT_FILE" ]]; then
    exit 0
fi

# Manual stop signal → graceful exit
if [[ -f "$STOP_FILE" ]]; then
    rm -f "$STOP_FILE"
    exit 0
fi

# Read checkpoint (fail gracefully)
if ! checkpoint=$(cat "$CHECKPOINT_FILE" 2>/dev/null); then
    exit 0
fi

# Parse fields — prefer jq, fallback to grep
HAS_JQ=false
command -v jq &>/dev/null && HAS_JQ=true

if $HAS_JQ; then
    status=$(echo "$checkpoint" | jq -r '.status // "unknown"')
    current_iteration=$(echo "$checkpoint" | jq -r '.current_iteration // 0')
    max_iterations=$(echo "$checkpoint" | jq -r '.max_iterations // 15')
    request=$(echo "$checkpoint" | jq -r '.request // ""')
else
    status=$(echo "$checkpoint" | grep -o '"status"[[:space:]]*:[[:space:]]*"[^"]*"' | cut -d'"' -f4 || echo "unknown")
    current_iteration=$(echo "$checkpoint" | grep -o '"current_iteration"[[:space:]]*:[[:space:]]*[0-9]*' | grep -o '[0-9]*$' || echo "0")
    max_iterations=$(echo "$checkpoint" | grep -o '"max_iterations"[[:space:]]*:[[:space:]]*[0-9]*' | grep -o '[0-9]*$' || echo "15")
    request=$(echo "$checkpoint" | grep -o '"request"[[:space:]]*:[[:space:]]*"[^"]*"' | cut -d'"' -f4 || echo "")
fi

# Already completed → allow exit
if [[ "$status" == "completed" ]] || [[ "$status" == "aborted" ]]; then
    exit 0
fi

# All ACs complete → run quality gate before marking completed
if $HAS_JQ; then
    ac_total_count=$(echo "$checkpoint" | jq '[.acceptance_criteria // [] | .[]] | length' 2>/dev/null || echo "0")
    ac_done_count=$(echo "$checkpoint" | jq '[.acceptance_criteria // [] | .[] | select(.done == true)] | length' 2>/dev/null || echo "0")
    if [[ "$ac_total_count" -gt 0 ]] && [[ "$ac_done_count" -ge "$ac_total_count" ]]; then
        # Quality gate: run build + test before allowing completion
        qg_failed=""
        repo_root=$(git rev-parse --show-toplevel 2>/dev/null || pwd)

        if [[ -f "$repo_root/package.json" ]]; then
            # TypeScript check
            if [[ -f "$repo_root/tsconfig.json" ]]; then
                if ! (cd "$repo_root" && npx tsc --noEmit 2>/dev/null); then
                    qg_failed="${qg_failed}TypeScript compilation failed. "
                fi
            fi
            # Test check
            if grep -q '"test"' "$repo_root/package.json" 2>/dev/null; then
                if ! (cd "$repo_root" && npm test --silent 2>/dev/null); then
                    qg_failed="${qg_failed}Tests failed. "
                fi
            fi
        fi

        if [[ -n "$qg_failed" ]]; then
            # Quality gate failed → block and ask Claude to fix
            qg_prompt="All ACs are marked done, but quality gate FAILED:
${qg_failed}

Fix the issues, then invoke /quality-gate to verify everything passes.
Do NOT set status to \"completed\" until quality gate passes."

            json_reason=$(echo "$qg_prompt" | python3 -c 'import json,sys; print(json.dumps(sys.stdin.read()))' 2>/dev/null) || {
                json_reason="\"Quality gate failed: ${qg_failed}\""
            }
            cat <&lt;QGEOF
{
  "decision": "block",
  "reason": $json_reason
}
QGEOF
            exit 0
        fi

        # Quality gate passed → mark completed and exit
        echo "$checkpoint" | jq '.status = "completed"' > "$CHECKPOINT_FILE"
        exit 0
    fi
fi

# Max iterations reached → update status and exit
if [[ "$current_iteration" -ge "$max_iterations" ]]; then
    if $HAS_JQ; then
        echo "$checkpoint" | jq '.status = "max_iterations_reached"' > "$CHECKPOINT_FILE"
    else
        echo "$checkpoint" | sed 's/"status"[[:space:]]*:[[:space:]]*"[^"]*"/"status": "max_iterations_reached"/' > "$CHECKPOINT_FILE"
    fi
    exit 0
fi

# Increment iteration
new_iteration=$((current_iteration + 1))
echo "$new_iteration" > "$ITERATION_FILE"

# Update checkpoint iteration
if $HAS_JQ; then
    echo "$checkpoint" | jq ".current_iteration = $new_iteration" > "$CHECKPOINT_FILE"
else
    echo "$checkpoint" | sed "s/\"current_iteration\"[[:space:]]*:[[:space:]]*[0-9]*/\"current_iteration\": $new_iteration/" > "$CHECKPOINT_FILE"
fi

# Extract AC status for prompt
if $HAS_JQ; then
    ac_status=$(echo "$checkpoint" | jq -r '
        .acceptance_criteria // [] | to_entries[] |
        (if .value.done then "[x]" else "[ ]" end) + " " + .value.description
    ' 2>/dev/null || echo "Check .auto-dev/checkpoint.json")
    ac_done=$(echo "$checkpoint" | jq '[.acceptance_criteria // [] | .[] | select(.done == true)] | length' 2>/dev/null || echo "0")
    ac_total=$(echo "$checkpoint" | jq '[.acceptance_criteria // [] | .[]] | length' 2>/dev/null || echo "0")
else
    ac_status="(Install jq for detailed AC status)"
    ac_done="?"
    ac_total="?"
fi

# Build TDD prompt for next iteration
# Deliberately references existing tdd-guide and quality-gate
tdd_prompt="Continue Auto-Dev iteration #$new_iteration / $max_iterations

Original request: $request

Acceptance Criteria ($ac_done/$ac_total complete):
$ac_status

Instructions for this iteration:
1. Pick the NEXT incomplete AC (marked [ ])
2. Follow TDD: RED (write failing test) → GREEN (minimal implementation) → REFACTOR
3. VALIDATE: Invoke /quality-gate skill (NOT manual tsc/vitest commands)
4. If quality-gate passes, update the AC in .auto-dev/checkpoint.json: set done=true
5. Process ONE AC only, then STOP to let the Stop Hook verify
6. When ALL ACs are done, invoke /code-review before setting status to \"completed\"

IMPORTANT: You MUST invoke /quality-gate — the Stop Hook will verify build+test independently."

# JSON-encode the prompt safely
json_reason=$(echo "$tdd_prompt" | python3 -c 'import json,sys; print(json.dumps(sys.stdin.read()))' 2>/dev/null) || {
    json_reason="\"Continue Auto-Dev iteration #$new_iteration / $max_iterations ($ac_done/$ac_total AC complete)\""
}

cat <&lt;EOF
{
  "decision": "block",
  "reason": $json_reason
}
EOF

#!/bin/bash
# bash-pre-check.sh — 合併 pre-commit-check + long-running-command-warning
# 只 fork 一次 jq，處理所有 Bash PreToolUse 檢查
#
# Called by Claude Code hooks system (PreToolUse: Bash)
# Stdin: JSON with tool_input.command

# Cross-platform timeout: macOS 用 gtimeout (coreutils), 無則直接執行
if command -v timeout &>/dev/null; then
  TIMEOUT_CMD="timeout 30"
elif command -v gtimeout &>/dev/null; then
  TIMEOUT_CMD="gtimeout 30"
else
  TIMEOUT_CMD=""
fi

INPUT_JSON=$(cat)
COMMAND=$(echo "$INPUT_JSON" | jq -r '.tool_input.command // ""' 2>/dev/null)

if [ -z "$COMMAND" ]; then
  exit 0
fi

# --- 1. Pre-commit check: git commit 時跑 npm run lint（與 CI 一致）---
if echo "$COMMAND" | grep -qE '^\s*git\s+commit\b'; then
  # 確認在 git repo 中
  if git rev-parse --is-inside-work-tree &>/dev/null; then
    STAGED_FILES=$(git diff --cached --name-only --diff-filter=ACM 2>/dev/null | grep -E '\.(js|jsx|ts|tsx|mjs|cjs)$' || true)

    if [ -n "$STAGED_FILES" ]; then
      REPO_ROOT=$(git rev-parse --show-toplevel 2>/dev/null)

      if [ -n "$REPO_ROOT" ] && [ -f "$REPO_ROOT/package.json" ]; then
        # 只在 package.json 有 lint script 時才跑（.NET 等專案自動跳過）
        if jq -e '.scripts.lint' "$REPO_ROOT/package.json" &>/dev/null; then
          echo "Running npm run lint (matching CI)..."
          (cd "$REPO_ROOT" && npm run lint) || {
            echo "✘ npm run lint failed — commit blocked. Fix lint errors before committing."
            exit 2
          }
        fi

        # --- Node.js test check ---
        TEST_SCRIPT=$(jq -r '.scripts.test // ""' "$REPO_ROOT/package.json" 2>/dev/null)
        if [ -n "$TEST_SCRIPT" ] && ! echo "$TEST_SCRIPT" | grep -q 'echo.*Error.*no test'; then
          echo "Running npm test (pre-commit)..."
          (cd "$REPO_ROOT" && $TIMEOUT_CMD npm test) || {
            echo "✘ npm test failed — commit blocked. Fix test failures before committing."
            exit 2
          }
        fi
      fi
    fi

    # --- Python test check ---
    STAGED_PY=$(git diff --cached --name-only --diff-filter=ACM 2>/dev/null | grep -E '\.py$' || true)
    if [ -n "$STAGED_PY" ]; then
      REPO_ROOT="${REPO_ROOT:-$(git rev-parse --show-toplevel 2>/dev/null)}"
      HAS_PYTEST=false
      if [ -f "$REPO_ROOT/pytest.ini" ] || { [ -f "$REPO_ROOT/setup.cfg" ] && grep -q '\[tool:pytest\]' "$REPO_ROOT/setup.cfg" 2>/dev/null; }; then
        HAS_PYTEST=true
      elif [ -f "$REPO_ROOT/pyproject.toml" ] && grep -q '\[tool\.pytest' "$REPO_ROOT/pyproject.toml" 2>/dev/null; then
        HAS_PYTEST=true
      fi
      if [ "$HAS_PYTEST" = true ]; then
        echo "Running pytest (pre-commit)..."
        PYTEST_CMD="pytest"
        if ! command -v pytest &>/dev/null && python3 -m pytest --version &>/dev/null; then
          PYTEST_CMD="python3 -m pytest"
        fi
        (cd "$REPO_ROOT" && $TIMEOUT_CMD $PYTEST_CMD) || {
          echo "✘ pytest failed — commit blocked. Fix test failures before committing."
          exit 2
        }
      fi
    fi

    # --- Go test check ---
    STAGED_GO=$(git diff --cached --name-only --diff-filter=ACM 2>/dev/null | grep -E '\.go$' || true)
    if [ -n "$STAGED_GO" ]; then
      REPO_ROOT="${REPO_ROOT:-$(git rev-parse --show-toplevel 2>/dev/null)}"
      if find "$REPO_ROOT" -name '*_test.go' -maxdepth 3 2>/dev/null | grep -q .; then
        echo "Running go test (pre-commit)..."
        (cd "$REPO_ROOT" && $TIMEOUT_CMD go test ./...) || {
          echo "✘ go test failed — commit blocked. Fix test failures before committing."
          exit 2
        }
      fi
    fi

    # --- Rust test check ---
    STAGED_RS=$(git diff --cached --name-only --diff-filter=ACM 2>/dev/null | grep -E '\.rs$' || true)
    if [ -n "$STAGED_RS" ]; then
      REPO_ROOT="${REPO_ROOT:-$(git rev-parse --show-toplevel 2>/dev/null)}"
      if [ -f "$REPO_ROOT/Cargo.toml" ]; then
        echo "Running cargo test (pre-commit)..."
        (cd "$REPO_ROOT" && $TIMEOUT_CMD cargo test) || {
          echo "✘ cargo test failed — commit blocked. Fix test failures before committing."
          exit 2
        }
      fi
    fi
  fi
  exit 0
fi

# --- 2. Long-running command warning ---
if echo "$COMMAND" | grep -qE '^(npm|pnpm|yarn|cargo|pip|go|make|docker|pytest|jest|vitest)\s+(run|install|build|test|start|dev|serve)'; then
  echo '⚠️ 長時間命令建議使用 tmux 或 run_in_background 執行'
fi

exit 0

git-strategy-stop.sh ​

#!/bin/bash
# Git Strategy Auto-Detection Stop Hook
# Detects git context at session end and blocks with strategy options
#
# Activation: all conditions must be met
#   - Inside a git repo
#   - Has code changes (tracked modified or untracked new files)
#   - Excludes config dirs (.claude/, .auto-dev/, etc.)
#   - auto-dev loop is NOT active
#
# Output: JSON {"decision": "block", "reason": "..."} or silent exit 0

set -euo pipefail

# ── Guard: CWD still exists (worktree may have been deleted) ──
[[ -d "${PWD}" ]] || exit 0

# ── Guard: not in a git repo ──
git rev-parse --is-inside-work-tree &>/dev/null || exit 0

# ── Guard: already asked in this session → skip (prevent block loop) ──
# Lock TTL: 10 minutes — auto-expires if session ends without committing
LOCK_FILE="/tmp/git-strategy-stop-$(echo "${PWD}" | sed 's/\//_/g').lock"
if [[ -f "${LOCK_FILE}" ]]; then
    LOCK_AGE=$(( $(date +%s) - $(stat -f%m "${LOCK_FILE}" 2>/dev/null || stat -c%Y "${LOCK_FILE}" 2>/dev/null || echo 0) ))
    if [[ "${LOCK_AGE}" -lt 600 ]]; then
        exit 0
    fi
    rm -f "${LOCK_FILE}"
fi

# ── Guard: auto-dev loop active → skip ──
if [[ -f ".auto-dev/checkpoint.json" ]]; then
    if command -v jq &>/dev/null; then
        ad_status=$(jq -r '.status // "unknown"' .auto-dev/checkpoint.json 2>/dev/null || echo "unknown")
    else
        ad_status=$(grep -o '"status"[[:space:]]*:[[:space:]]*"[^"]*"' .auto-dev/checkpoint.json 2>/dev/null | cut -d'"' -f4 || echo "unknown")
    fi
    [[ "${ad_status}" == "in_progress" ]] && exit 0
fi

# ── Detect code changes (exclude config directories) ──
CODE_PATTERN='\.(ts|js|vue|tsx|jsx|py|go|rs|rb|java|kt|swift|c|cpp|h|css|scss|html)$'

# Tracked modified files (staged + unstaged), excluding config dirs
MODIFIED=$(git diff --name-only HEAD 2>/dev/null | grep -vE '^\.(claude|auto-dev)/' | grep -E "${CODE_PATTERN}" || true)

# Untracked new code files, excluding config dirs
UNTRACKED=$(git ls-files --others --exclude-standard 2>/dev/null | grep -vE '^\.(claude|auto-dev)/' | grep -E "${CODE_PATTERN}" || true)

# No code changes → silent exit
if [[ -z "${MODIFIED}" ]] && [[ -z "${UNTRACKED}" ]]; then
    exit 0
fi

# ── Gather git context ──
BRANCH=$(git branch --show-current 2>/dev/null || echo "detached")

# Protected branch?
PROTECTED=false
case "${BRANCH}" in main|master|staging|develop) PROTECTED=true ;; esac

# Worktree detection: .git is a file (not dir) in worktrees
IS_WORKTREE=false
TOPLEVEL=$(git rev-parse --show-toplevel 2>/dev/null || echo ".")
[[ -f "${TOPLEVEL}/.git" ]] && IS_WORKTREE=true

# Remote tracking?
HAS_REMOTE=false
git rev-parse --abbrev-ref "@{upstream}" &>/dev/null && HAS_REMOTE=true

# Change summary
CHANGES=$(git diff --stat HEAD 2>/dev/null | tail -1 || echo "unknown")

# Count untracked code files (avoid double-output from grep -c || echo)
if [[ -n "${UNTRACKED}" ]]; then
    UNTRACKED_COUNT=$(echo "${UNTRACKED}" | wc -l | tr -d ' ')
else
    UNTRACKED_COUNT=0
fi

# ── Build strategy message based on mode ──

if ${IS_WORKTREE}; then
    # ── Mode C: Worktree ──
    WORKTREE_PATH=$(git rev-parse --show-toplevel 2>/dev/null || echo "unknown")
    # Find the main branch by checking common names
    MAIN_BRANCH=""
    for candidate in main master; do
        if git show-ref --verify --quiet "refs/heads/${candidate}" 2>/dev/null; then
            MAIN_BRANCH="${candidate}"
            break
        fi
    done
    [[ -z "${MAIN_BRANCH}" ]] && MAIN_BRANCH="main"

    STRATEGY="Git 策略確認 (worktree: ${WORKTREE_PATH})
分支: ${BRANCH}
主分支: ${MAIN_BRANCH}
變更: ${CHANGES}"
    [[ "${UNTRACKED_COUNT}" -gt 0 ]] && STRATEGY="${STRATEGY}
新增檔案: ${UNTRACKED_COUNT} 個未追蹤的程式碼檔案"
    STRATEGY="${STRATEGY}

選項:
  1. commit -> push -> 開 PR 合併回 ${MAIN_BRANCH}
  2. commit -> 直接 merge 回 ${MAIN_BRANCH}
  3. 只 commit (繼續在 worktree 開發)
  4. 暫不處理 (稍後再決定)

請問要採用哪個策略?

⚠️ 請使用 AskUserQuestion 工具呈現以上選項給用戶，勿直接用文字回覆。"

elif ${PROTECTED}; then
    # ── Mode A: Protected branch ──
    STRATEGY="Git 策略確認 (保護分支: ${BRANCH})
變更: ${CHANGES}"
    [[ "${UNTRACKED_COUNT}" -gt 0 ]] && STRATEGY="${STRATEGY}
新增檔案: ${UNTRACKED_COUNT} 個未追蹤的程式碼檔案"
    STRATEGY="${STRATEGY}

選項:
  1. 建立新分支 -> commit -> 開 PR
  2. 直接 commit 到 ${BRANCH} (小修改)
  3. 暫不處理 (稍後再決定)

請問要採用哪個策略?

⚠️ 請使用 AskUserQuestion 工具呈現以上選項給用戶，勿直接用文字回覆。"

else
    # ── Mode B: Feature branch ──
    STRATEGY="Git 策略確認 (${BRANCH})
變更: ${CHANGES}"
    [[ "${UNTRACKED_COUNT}" -gt 0 ]] && STRATEGY="${STRATEGY}
新增檔案: ${UNTRACKED_COUNT} 個未追蹤的程式碼檔案"
    ${HAS_REMOTE} && STRATEGY="${STRATEGY}
Remote: origin/${BRANCH} (tracked)"
    STRATEGY="${STRATEGY}

選項:
  1. commit -> push (觸發 CI/CD)
  2. 只 commit 不 push
  3. 暫不處理 (稍後再決定)

請問要採用哪個策略?

⚠️ 請使用 AskUserQuestion 工具呈現以上選項給用戶，勿直接用文字回覆。"

fi

# ── Create lock to prevent re-triggering ──
touch "${LOCK_FILE}"

# ── Output block decision ──
json_reason=$(echo "${STRATEGY}" | python3 -c 'import json,sys; print(json.dumps(sys.stdin.read()))' 2>/dev/null) || {
    # Fallback: simple single-line message
    json_reason="\"Git 策略確認: 分支 ${BRANCH} 有未提交的程式碼變更，請確認 git 策略\""
}

cat <&lt;EOF
{
  "decision": "block",
  "reason": ${json_reason}
}
EOF

#!/usr/bin/env bash
# log-skill.sh — PreToolUse hook for Skill tool
# Logs skill invocations to ~/.claude/skill-usage.log

#!/usr/bin/env bash
# session-sync.sh — SessionStart 統一同步腳本
# 合併原本分散的三個同步機制，每段獨立執行，互不影響
# 不使用 set -e，確保某段失敗不阻止其他段

# ── Section 1: Fix hookify plugin Python import ──
# hookify 的 Python hook 使用 `from hookify.core.config_loader import ...`
# 但 plugin cache 結構為 hookify/&lt;version>/core/...，版本目錄導致找不到 package
# 在每個版本目錄內建立 hookify -> . 的 self-symlink 解決
CACHE_BASE="$HOME/.claude/plugins/cache/claude-code-plugins/hookify"
if [ -d "$CACHE_BASE" ]; then
  for version_dir in "$CACHE_BASE"/*/; do
    [ -d "$version_dir" ] || continue
    symlink_path="${version_dir}hookify"
    if [ ! -e "$symlink_path" ]; then
      ln -s . "$symlink_path" 2>/dev/null
    fi
  done
fi

# ── Section 2: Sync dotfiles skills ──
# 遍歷 ~/dotfiles/claude/skills/*/，若 ~/.claude/skills/ 缺少對應 symlink 則建立
# 跳過 learned/ 目錄
DOTFILES_SKILLS="$HOME/dotfiles/claude/skills"
TARGET_SKILLS="$HOME/.claude/skills"
if [ -d "$DOTFILES_SKILLS" ]; then
  mkdir -p "$TARGET_SKILLS"
  for skill_dir in "$DOTFILES_SKILLS"/*/; do
    [ -d "$skill_dir" ] || continue
    name=$(basename "$skill_dir")
    [ "$name" = "learned" ] && continue
    dst="$TARGET_SKILLS/$name"
    if [ ! -L "$dst" ] && [ ! -e "$dst" ]; then
      ln -s "$skill_dir" "$dst" 2>/dev/null
    fi
  done
fi

# ── Section 3: Sync dotfiles hookify rules ──
# 遍歷 ~/dotfiles/claude/hookify.*.md，若 ~/.claude/ 缺少對應 symlink 則建立
for f in "$HOME/dotfiles/claude"/hookify.*.md; do
  [ -f "$f" ] || continue
  name=$(basename "$f")
  dst="$HOME/.claude/$name"
  if [ ! -L "$dst" ] && [ ! -e "$dst" ]; then
    ln -s "$f" "$dst" 2>/dev/null
  fi
done

exit 0

#!/bin/bash
# stop-orchestrator.sh — 統一 Stop Hook 協調器
#
# 按優先級執行所有自訂 Stop Hook：
#   1. auto-dev-stop.sh   — 僅 TDD 模式（.auto-dev/checkpoint.json 存在且 in_progress）
#   2. git-strategy-stop.sh — 一般模式（非 TDD 時執行）
#
# 功能：
#   - 統一 lock：防止 git-strategy 在同一 session 重複觸發（TTL 10 分鐘）
#   - 超時保護：每個子 hook 最多 30 秒，超時跳過並記錄
#   - 執行日誌：記錄到 ~/.claude/stop-log/YYYYMMDD-HHMMSS.log
#
# 輸出格式（若需 block）：
#   {"decision": "block", "reason": "..."}

# 不用 set -e：要手動處理各子 hook 的退出碼
set -uo pipefail

HOOKS_DIR="$HOME/.claude/hooks"
LOG_DIR="$HOME/.claude/stop-log"
HOOK_TIMEOUT=30
LOCK_TTL=600  # 10 分鐘

# ── 初始化日誌目錄 ──────────────────────────────────
mkdir -p "$LOG_DIR"
TIMESTAMP=$(date '+%Y%m%d-%H%M%S')
LOG_FILE="${LOG_DIR}/${TIMESTAMP}.log"

log() {
    echo "[$(date '+%H:%M:%S')] $*" >> "$LOG_FILE" 2>/dev/null || true
}

# ── Lock file：基於 PWD 做 scoped 鎖 ──────────────────
# 使用 cksum 取得跨平台可用的 hash（macOS/Linux 皆有）
SESSION_HASH=$(printf '%s' "${PWD}" | cksum | awk '{print $1}')
LOCK_FILE="/tmp/stop-orchestrator-${SESSION_HASH}.lock"

# ── HOOK_OUTPUT：子 hook 執行結果（透過全域變數傳回）──
HOOK_OUTPUT=""

# ── run_hook: 執行子 hook，timeout 30s，結果存 HOOK_OUTPUT ──
# 引數：$1 = 名稱（for log）, $2 = script 路徑
run_hook() {
    local name="$1"
    local script="$2"
    local exit_code=0
    HOOK_OUTPUT=""

    log "START ${name}"

    # Guard: script 不存在
    if [[ ! -f "${script}" ]]; then
        log "SKIP ${name} (script not found: ${script})"
        return 0
    fi

    # Guard: script 不可執行時，補上執行權
    if [[ ! -x "${script}" ]]; then
        chmod +x "${script}" 2>/dev/null || true
    fi

    # 執行並捕捉 stdout；stderr 導向 log
    HOOK_OUTPUT=$(timeout "${HOOK_TIMEOUT}" bash "${script}" 2>>"${LOG_FILE}") || exit_code=$?

    if [[ "${exit_code}" -eq 124 ]]; then
        log "TIMEOUT ${name} (exceeded ${HOOK_TIMEOUT}s)"
        HOOK_OUTPUT=""
        return 0
    fi

    log "DONE ${name} (exit=${exit_code}, output_bytes=${#HOOK_OUTPUT})"
    return "${exit_code}"
}

# ── is_block: 判斷子 hook 輸出是否為 block 決策 ──
# 回傳 0 = block，1 = 不 block
# 透過 stdin 傳入 JSON，避免引號逸脫問題
is_block() {
    local output="$1"
    [[ -z "${output}" ]] && return 1
    printf '%s' "${output}" | python3 -c '
import json, sys
try:
    d = json.load(sys.stdin)
    sys.exit(0 if d.get("decision") == "block" else 1)
except Exception:
    sys.exit(1)
' 2>/dev/null
}

log "=== ORCHESTRATOR START (PWD=${PWD}) ==="

# ═══════════════════════════════════════════════════
# TDD 模式偵測（不依賴環境變數，避免 subshell 遺失）
# ═══════════════════════════════════════════════════
AUTODEV_ACTIVE=false
AUTODEV_CHECKPOINT=".auto-dev/checkpoint.json"

if [[ -f "${AUTODEV_CHECKPOINT}" ]]; then
    if command -v jq &>/dev/null; then
        ad_status=$(jq -r '.status // "unknown"' "${AUTODEV_CHECKPOINT}" 2>/dev/null || echo "unknown")
    else
        # Fallback: grep-based 解析
        ad_status=$(grep -o '"status"[[:space:]]*:[[:space:]]*"[^"]*"' "${AUTODEV_CHECKPOINT}" 2>/dev/null \
            | head -1 | sed 's/.*"\([^"]*\)"$/\1/' || echo "unknown")
    fi
    [[ "${ad_status}" == "in_progress" ]] && AUTODEV_ACTIVE=true
    log "TDD mode: status=${ad_status}, active=${AUTODEV_ACTIVE}"
else
    log "TDD mode: no checkpoint found"
fi

# ═══════════════════════════════════════════════════
# Priority 2: auto-dev-stop — TDD 模式時執行（不受 lock 限制）
# ═══════════════════════════════════════════════════
if ${AUTODEV_ACTIVE}; then
    run_hook "auto-dev-stop" "${HOOKS_DIR}/auto-dev-stop.sh" || true

    if is_block "${HOOK_OUTPUT}"; then
        log "BLOCK by auto-dev-stop"
        log "=== ORCHESTRATOR END (blocked by auto-dev) ==="
        printf '%s\n' "${HOOK_OUTPUT}"
        exit 0
    fi

    log "=== ORCHESTRATOR END (TDD mode, no block) ==="
    exit 0
fi

# ═══════════════════════════════════════════════════
# Lock check（僅非 TDD 模式適用）
# 防止 git-strategy 在同一 session 中重複提問
# 使用 mkdir 原子操作：mkdir 失敗代表 lock 已存在（POSIX 保證原子性）
# ═══════════════════════════════════════════════════
if [[ -d "${LOCK_FILE}" ]]; then
    lock_mtime=$(stat -f%m "${LOCK_FILE}" 2>/dev/null \
        || stat -c%Y "${LOCK_FILE}" 2>/dev/null \
        || echo 0)
    lock_age=$(( $(date +%s) - lock_mtime ))

    if [[ "${lock_age}" -lt "${LOCK_TTL}" ]]; then
        log "LOCKED (age=${lock_age}s < TTL=${LOCK_TTL}s), skip git-strategy"
        log "=== ORCHESTRATOR END (locked) ==="
        exit 0
    fi

    # Lock 過期，清除
    rm -rf "${LOCK_FILE}"
    log "LOCK expired (age=${lock_age}s), removed"
fi

# ═══════════════════════════════════════════════════
# Priority 3: git-strategy-stop — 一般模式
# ═══════════════════════════════════════════════════
run_hook "git-strategy-stop" "${HOOKS_DIR}/git-strategy-stop.sh" || true

if is_block "${HOOK_OUTPUT}"; then
    log "BLOCK by git-strategy-stop"
    # 設置 orchestrator lock（mkdir 原子操作），防止 TTL 內重複觸發
    mkdir "${LOCK_FILE}" 2>/dev/null || true
    log "LOCK set (${LOCK_FILE})"
    log "=== ORCHESTRATOR END (blocked by git-strategy) ==="
    printf '%s\n' "${HOOK_OUTPUT}"
    exit 0
fi

log "=== ORCHESTRATOR END (ok, no block) ==="
exit 0

#!/bin/bash
# Strategic Compact Suggester
# Runs on PreToolUse to suggest manual compaction at logical intervals
#
# Why manual over auto-compact:
# - Auto-compact happens at arbitrary points, often mid-task
# - Strategic compacting preserves context through logical phases
# - Compact after exploration, before execution
# - Compact after completing a milestone, before starting next

# Read session_id from stdin JSON payload (stable across hook invocations)
INPUT=$(cat)
SESSION_ID=$(echo "$INPUT" | jq -r '.session_id // empty')
if [ -z "$SESSION_ID" ]; then
  exit 0
fi

COUNTER_FILE="/tmp/claude-strategic-compact-${SESSION_ID}"
THRESHOLD=${COMPACT_THRESHOLD:-50}

# Increment counter (default 0 if file missing)
count=$(( $(cat "$COUNTER_FILE" 2>/dev/null || echo 0) + 1 ))
echo "$count" > "$COUNTER_FILE"

# First suggestion at threshold
if [ "$count" -eq "$THRESHOLD" ]; then
  echo "[StrategicCompact] ${count} tool calls reached — consider running /compact if you're transitioning between phases (e.g., exploration → implementation, or between milestones)"
fi

# Periodic reminders every 25 calls after threshold
past_threshold=$(( count - THRESHOLD ))
if [ "$count" -gt "$THRESHOLD" ] && [ $(( past_threshold % 25 )) -eq 0 ]; then
  echo "[StrategicCompact] ${count} tool calls — good checkpoint for /compact if context is stale"
fi
